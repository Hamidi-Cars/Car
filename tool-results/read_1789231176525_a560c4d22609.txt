    28→/** فشرده‌سازی عکس انتخاب‌شده در مرورگر (حداکثر عرض ۱۲۸۰px، JPEG ۸۵٪) */
    29→function fileToDataUrl(file: File): Promise<string> {
    30→  return new Promise((resolve, reject) => {
    31→    const reader = new FileReader();
    32→    reader.onerror = () => reject(new Error("خطا در خواندن فایل"));
    33→    reader.onload = () => {
    34→      const img = new Image();
    35→      img.onerror = () => reject(new Error("فایل تصویر معتبر نیست"));
    36→      img.onload = () => {
    37→        const maxW = 1280;
    38→        const scale = Math.min(1, maxW / img.width);
    39→        const w = Math.round(img.width * scale);
    40→        const h = Math.round(img.height * scale);
    41→        const canvas = document.createElement("canvas");
    42→        canvas.width = w;
    43→        canvas.height = h;
    44→        const ctx = canvas.getContext("2d");
    45→        if (!ctx) return reject(new Error("خطا در پردازش تصویر"));
    46→        ctx.drawImage(img, 0, 0, w, h);
    47→        resolve(canvas.toDataURL("image/jpeg", 0.85));
    48→      };
    49→      img.src = reader.result as string;
    50→    };
    51→    reader.readAsDataURL(file);
    52→  });
    53→}
